# UI Rework Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development
> (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use
> checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the web app feel alive: a zoomable, hoverable, animated map with a new Hive
(honeycomb) opening view and geometric agent marks, a legend dock, a richer side panel, a calmer
rail with an account menu, a shared motion language on every page, and a Hive theme as the
default.

**Architecture:** Everything is client state over the existing map payload. New pure modules
(`map/hive.ts`, `map/camera.ts`, `map/zoomLabels.ts`, `map/stagger.ts`) carry the maths and are
unit-tested; React components wire them. The map keeps its DOM dots (real buttons) inside one
transform layer (`world`). The theme pipeline gains a Hive theme and one `--mark` token. The
member's name, role and hive load once in the root layout, alongside every page's own loads.

**Tech Stack:** Next.js 16 (App Router), React 19, TypeScript, CSS modules, vitest 5, Playwright.
No new runtime dependency.

**Spec:** `docs/superpowers/specs/2026-11-23-ui-rework-design.md`. Read it before any task. The
approved visual reference is the brainstorm prototype
(`.superpowers/brainstorm/*/content/map-prototype-hive-marks.html`, not committed); the spec
wins where they differ.

## Global Constraints

- Work only in the worktree `.claude/worktrees/ui-rework` on branch `task/ui-rework`.
- Stage explicitly by path (`git add <paths>`); never `git add -A` / `.` / `commit -a`.
- Commits use Conventional Commits with a **lower-case subject** (commitlint rejects
  sentence-case), e.g. `feat(map): add the camera`. End every message with
  `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- No new runtime dependency. No change under `supabase/` and no new migration (nothing here
  changes who can see what).
- Every interaction is client state only: **0 server round-trips** for hover, zoom, pan, fit,
  view switch, dock preview/pin, agents card, account menu and panel changes.
- Easing `--ease: cubic-bezier(0.16, 1, 0.3, 1)`; durations `--dur-fast` 0.18s, `--dur` 0.3s,
  `--dur-slow` 0.6s.
- `prefers-reduced-motion: reduce` stops every animation and transition added here (glides,
  stagger, pulses, glow, marks moving, count-ups, rises, camera glides become instant).
- Hairlines brighten on hover, never thicken; no drop shadows on controls (ADR 13).
- Never paint colours by theme name: a component uses tokens only (`no-literals` tests enforce
  this in `map/` and `app/`).
- Text read from the hive is rendered as text nodes only (titles, names); never
  `dangerouslySetInnerHTML`.
- Copy: sentence case, plain verbs. Exact strings in this plan are the strings to ship.
- Gates before done: `pnpm typecheck && pnpm lint && pnpm test && pnpm build` from the
  worktree root, plus `pnpm --filter @hive-mind/web test:e2e`.
- Run web unit tests with `pnpm --filter @hive-mind/web exec vitest run <file>`; e2e with

// web/src/shell/useSlidingPill.ts
"use client";
// A highlight that slides under the active control of a segmented group. The group stays the
// accessible thing (buttons with aria-pressed, or tabs); the pill is decoration (aria-hidden).
import {
  useLayoutEffect,
  useState,
  type CSSProperties,
  type RefObject,
} from "react";

export function pillRect(
  container: DOMRect,
  active: DOMRect,
): { left: number; width: number } {
  return {
    left: active.left - container.left,
    width: Math.max(0, active.width),
  };
}

export function useSlidingPill(
  group: RefObject<HTMLElement | null>,
  activeSelector: string,
  key: unknown,
): CSSProperties {
  const [style, setStyle] = useState<CSSProperties>({ opacity: 0 });
  useLayoutEffect(() => {
    const el = group.current;
    if (!el) return;
    const measure = () => {
      const active = el.querySelector<HTMLElement>(activeSelector);
      if (!active) return setStyle({ opacity: 0 });
      const { left, width } = pillRect(
        el.getBoundingClientRect(),
        active.getBoundingClientRect(),
      );
      setStyle({ left, width, opacity: 1 });
    };
    measure();
    const ro = new ResizeObserver(measure);
    ro.observe(el);
    return () => ro.disconnect();
  }, [group, activeSelector, key]);
  return style;
}
```

export type Axial = [number, number]; // [q, r]
export type CellKind = "root" | "hub" | "item" | "empty";
export type Cell = {
  q: number;
  r: number;
  x: number;
  y: number;
  kind: CellKind;
  node: number | null; // model node index, null for empty cells
  topic: number; // -1 for root and empty
  honey: number; // fill opacity 0..1 (see honey())
  capped: boolean; // locked item
};
export type MarkState = "live" | "today" | "idle";
export type AgentMark = {
  node: number; // agent node index
  state: MarkState;
  at: number | null; // item node index it sits on (null when idle)
  pos: [number, number]; // where the mark is drawn
};
export type Comb = {
  radius: number; // grid radius in cells (>= 12)
  size: number; // cell circumradius in box units
  cells: Cell[];
  pos: [number, number][]; // one per model node (agents included)
  axial: Map<number, Axial>; // node index -> its cell
  entrance: [number, number]; // centre of the entrance arch under the comb
  marks: AgentMark[];
};
export const HIVE_MIN_RADIUS = 12;
export const HIVE_CELL = 15.5;
export function honey(hoursSinceChange: number): number; // max(0.06, 0.5 * exp(-h / 336))
export function hexDist(a: Axial, b: Axial): number;
export function gridRadiusFor(items: number): number; // smallest R >= 12 with 3R^2+3R+1 >= ceil(items * 1.15)
export function markState(lastUsedAt: string | null, now: number): MarkState;
export function hiveComb(
  model: MapModel,
  now: number,
  agentsLastUsed: Map<string, string | null>, // agent node id -> last_used_at
): Comb;
/** A live mark's next cell: another item of the same member, changed within 120 h, at most 3 cells away; null if none. `pick` is in [0, 1). */
export function nextCell(
  comb: Comb,
  model: MapModel,
  mark: AgentMark,
  now: number,
  pick: number,
): number | null;
```

// The Hive view: one honeycomb. Each topic claims a contiguous territory of hexagonal cells
// around its hub; items sit nearest the hub, most recently changed first. Agents are geometric
// marks on their member's freshest cell (live, today) or at the entrance under the comb (idle).
// Pure: the same model and `now` always give the same comb.
import type { MapModel } from "./types";

export type Axial = [number, number];
export type CellKind = "root" | "hub" | "item" | "empty";
export type Cell = {
  q: number;
  r: number;
  x: number;
  y: number;
  kind: CellKind;
  node: number | null;
  topic: number;
  honey: number;
  capped: boolean;
};
export type MarkState = "live" | "today" | "idle";
export type AgentMark = {
  node: number;
  state: MarkState;
  at: number | null;
  pos: [number, number];
};
export type Comb = {
  radius: number;
  size: number;
  cells: Cell[];
  pos: [number, number][];
  axial: Map<number, Axial>;
  entrance: [number, number];
  marks: AgentMark[];
};

export const HIVE_MIN_RADIUS = 12;
export const HIVE_CELL = 15.5;
const CX = 500;
const CY = 380;
const SQ3 = Math.sqrt(3);
const HOUR = 3_600_000;
const LIVE_MS = HOUR; // matches agentStatus's "connected" window; keep in step with graph.ts
const TODAY_MS = 24 * HOUR;
const NEIGHBOURS: Axial[] = [
  [1, 0],
  [1, -1],
  [0, -1],
  [-1, 0],
  [-1, 1],
  [0, 1],
];
const key = (a: Axial) => `${a[0]},${a[1]}`;

export function hexDist(a: Axial, b: Axial): number {
  return (
    (Math.abs(a[0] - b[0]) +
      Math.abs(a[0] + a[1] - b[0] - b[1]) +
      Math.abs(a[1] - b[1])) /
    2
  );
}

export function honey(hours: number): number {
  return Math.max(0.06, 0.5 * Math.exp(-Math.max(0, hours) / 336));
}

export function gridRadiusFor(items: number): number {
  const need = Math.ceil(items * 1.15);
  let R = HIVE_MIN_RADIUS;
  while (3 * R * R + 3 * R + 1 < need) R++;
  return R;
}

export function markState(lastUsedAt: string | null, now: number): MarkState {
  if (lastUsedAt === null) return "idle";
  const t = Date.parse(lastUsedAt);
  if (!Number.isFinite(t)) return "idle";
  if (now - t <= LIVE_MS) return "live";
  if (now - t <= TODAY_MS) return "today";
  return "idle";
}

/** The comb's cell size: the minimum at radius 12, smaller as the grid grows, so it fits 740 tall. */
function cellSize(R: number): number {
  const fit = 300 / (1.5 * R + 1); // half-height of the comb within the box, with room for the entrance
  return Math.min(HIVE_CELL, fit);
}

function axialRound(qf: number, rf: number): Axial {
  const sf = -qf - rf;
  let q = Math.round(qf);
  let r = Math.round(rf);
  const s = Math.round(sf);
  const dq = Math.abs(q - qf);
  const dr = Math.abs(r - rf);
  const ds = Math.abs(s - sf);
  if (dq > dr && dq > ds) q = -r - s;
  else if (dr > ds) r = -q - s;
  return [q, r];
}

export function hiveComb(
  model: MapModel,
  now: number,
  agentsLastUsed: Map<string, string | null>,
): Comb {
  const items = model.nodes.filter(
    (n) => n.kind === "note" || n.kind === "file",
  );
  const R = gridRadiusFor(items.length + model.topics.length + 1);
  const S = cellSize(R);
  const xy = (a: Axial): [number, number] => [
    CX + S * SQ3 * (a[0] + a[1] / 2),
    CY + S * 1.5 * a[1],
  ];

  const all: Axial[] = [];
  for (let q = -R; q <= R; q++)
    for (let r = -R; r <= R; r++)
      if (hexDist([q, r], [0, 0]) <= R) all.push([q, r]);

  const T = model.topics.length;
  const owner = new Map<string, number | "root">([["0,0", "root"]]);
  const seedRing = R * 0.525; // 6.3 cells at R = 12
  const seeds: Axial[] = model.topics.map((_, i) => {
    const a = (i / Math.max(1, T)) * Math.PI * 2 - Math.PI / 2;
    const px = Math.cos(a) * seedRing * SQ3;
    const py = Math.sin(a) * seedRing * SQ3;
    return axialRound((SQ3 / 3) * px - py / 3, (2 / 3) * py);
  });

  const hours = (t: number) => (now - t) / HOUR;
  const members = model.topics.map((_, t) =>
    items
      .filter((n) => n.topic === t)
      .sort((a, b) => b.updatedAt - a.updatedAt || a.i - b.i),
  );
  const quota = members.map((m) => m.length + 1);
  const region: Axial[][] = model.topics.map(() => []);
  seeds.forEach((c, i) => {
    // Two seeds can round to the same cell only with many tiny topics; nudge outward.
    let s = c;
    while (owner.has(key(s))) s = [s[0], s[1] + 1];
    seeds[i] = s;
    owner.set(key(s), i);
    region[i]!.push(s);
    quota[i]!--;
  });

  let grew = true;
  while (grew) {
    grew = false;
    for (let i = 0; i < T; i++) {
      if (quota[i]! <= 0) continue;
      let best: Axial | null = null;
      let bd = Infinity;
      for (const c of region[i]!)
        for (const [dq, dr] of NEIGHBOURS) {
          const nb: Axial = [c[0] + dq, c[1] + dr];
          if (hexDist(nb, [0, 0]) > R || owner.has(key(nb))) continue;
          const d = hexDist(nb, seeds[i]!) + hexDist(nb, [0, 0]) * 0.08;
          if (d < bd || (d === bd && best && key(nb) < key(best))) {
            bd = d;
            best = nb;
          }
        }
      if (best) {
        owner.set(key(best), i);
        region[i]!.push(best);
        quota[i]!--;
        grew = true;
      }
    }
  }

  const pos: [number, number][] = model.nodes.map(() => [CX, CY]);
  const axial = new Map<number, Axial>([[0, [0, 0]]]);
  const cellNode = new Map<string, number>([["0,0", 0]]);
  for (let t = 0; t < T; t++) {
    const cs = region[t]!.slice().sort(
      (a, b) =>
        hexDist(a, seeds[t]!) - hexDist(b, seeds[t]!) ||
        key(a).localeCompare(key(b)),
    );
    const hub = model.nodes.find((n) => n.kind === "hub" && n.topic === t);
    if (hub && cs[0]) {
      pos[hub.i] = xy(cs[0]);
      axial.set(hub.i, cs[0]);
      cellNode.set(key(cs[0]), hub.i);
    }
    members[t]!.forEach((n, j) => {
      const c = cs[j + 1];
      if (!c) return;
      pos[n.i] = xy(c);
      axial.set(n.i, c);
      cellNode.set(key(c), n.i);
    });
  }

  const cells: Cell[] = all.map((a) => {
    const [x, y] = xy(a);
    const o = owner.get(key(a));
    const node = cellNode.get(key(a)) ?? null;
    if (o === "root")
      return {
        q: a[0],
        r: a[1],
        x,
        y,
        kind: "root",
        node: 0,
        topic: -1,
        honey: 0.12,
        capped: false,
      };
    if (o === undefined || node === null)
      return {
        q: a[0],
        r: a[1],
        x,
        y,
        kind: "empty",
        node: null,
        topic: -1,
        honey: 0,
        capped: false,
      };
    const n = model.nodes[node]!;
    const isHub = n.kind === "hub";
    return {
      q: a[0],
      r: a[1],
      x,
      y,
      kind: isHub ? "hub" : "item",
      node,
      topic: o,
      honey: isHub ? 0.32 : honey(hours(n.updatedAt)),
      capped: n.locked,
    };
  });

  const entrance: [number, number] = [CX, CY + S * 1.5 * R + 26];
  const agents = model.nodes.filter((n) => n.kind === "agent");
  const marks: AgentMark[] = agents.map((n) => {
    const state = markState(agentsLastUsed.get(n.id ?? "") ?? null, now);
    const fresh = items
      .filter((m) => m.member === n.member && axial.has(m.i))
      .sort((a, b) => b.updatedAt - a.updatedAt || a.i - b.i);
    const at = state === "idle" ? null : (fresh[0]?.i ?? null);
    return { node: n.i, state: at === null ? "idle" : state, at, pos: [0, 0] };
  });
  const idle = marks.filter((m) => m.state === "idle");
  idle.forEach((m, j) => {
    m.pos = [CX + (j - (idle.length - 1) / 2) * 18, entrance[1]];
  });
  for (const m of marks)
    if (m.at !== null) {
      const [x, y] = pos[m.at]!;
      m.pos = [x + S * 0.45, y - S * 0.52]; // on the cell's upper-right edge
    }
  for (const m of marks) pos[m.node] = m.pos;

  return { radius: R, size: S, cells, pos, axial, entrance, marks };
}

export function nextCell(
  comb: Comb,
  model: MapModel,
  mark: AgentMark,
  now: number,
  pick: number,
): number | null {
  if (mark.state !== "live" || mark.at === null) return null;
  const from = comb.axial.get(mark.at);
  if (!from) return null;
  const member = model.nodes[mark.node]!.member;
  const near = model.nodes.filter(
    (n) =>
      (n.kind === "note" || n.kind === "file") &&
      n.member === member &&
      n.i !== mark.at &&
      now - n.updatedAt <= 120 * HOUR &&
      comb.axial.has(n.i) &&
      hexDist(comb.axial.get(n.i)!, from) <= 3,
  );
  if (!near.length) return null;
  return near[Math.min(near.length - 1, Math.floor(pick * near.length))]!.i;
}

// web/src/map/camera.ts
// The map's camera: one transform (translate tx, ty, then scale k) over a world of w x h
// pixels that shows the 1000 x 760 box. Pure maths; the hook in useCamera.ts drives it.
export type Cam = { k: number; tx: number; ty: number };
export const K_MIN = 0.6;
export const K_MAX = 8;

export const clampK = (k: number): number =>
  Math.max(K_MIN, Math.min(K_MAX, k));

export function zoomAt(c: Cam, factor: number, px: number, py: number): Cam {
  const k = clampK(c.k * factor);
  return {
    k,
    tx: px - ((px - c.tx) * k) / c.k,
    ty: py - ((py - c.ty) * k) / c.k,
  };
}

export function wheelFactor(deltaY: number, ctrlKey: boolean): number {
  return Math.exp(-deltaY * (ctrlKey ? 0.01 : 0.0025));
}

export function fit(w: number, h: number, sw: number, sh: number): Cam {
  return { k: 1, tx: (sw - w) / 2, ty: (sh - h) / 2 };
}

export function toScreen(
  c: Cam,
  x: number,
  y: number,
  w: number,
  h: number,
): [number, number] {
  return [(x / 1000) * w * c.k + c.tx, (y / 760) * h * c.k + c.ty];
}

export function frame(
  points: [number, number][],
  w: number,
  h: number,
  sw: number,
  sh: number,
): Cam {
  const xs = points.map((p) => (p[0] / 1000) * w);
  const ys = points.map((p) => (p[1] / 760) * h);
  const x0 = Math.min(...xs);
  const x1 = Math.max(...xs);
  const y0 = Math.min(...ys);
  const y1 = Math.max(...ys);
  const k = Math.max(
    1,
    Math.min(3, sw / (x1 - x0 + 120), sh / (y1 - y0 + 120)),
  );
  const cx = (x0 + x1) / 2;
  const cy = (y0 + y1) / 2;
  return { k, tx: sw / 2 - cx * k, ty: sh / 2 - cy * k };
}

// web/src/map/zoomLabels.ts
// Semantic zoom: which lit dots get a label because the map is zoomed in (spec 1.2). Runs once a
// pan or zoom settles, never per frame.
export type LabelInput = {
  i: number;
  on: boolean;
  links: number;
  x: number;
  y: number;
  width: number;
};

type Box = { x0: number; y0: number; x1: number; y1: number };
const overlaps = (a: Box, b: Box) =>
  a.x0 < b.x1 && b.x0 < a.x1 && a.y0 < b.y1 && b.y0 < a.y1;

export function zoomLabels(
  dots: LabelInput[],
  k: number,
  stage: { w: number; h: number },
  screen: (x: number, y: number) => [number, number],
  cap = 60,
): Set<number> {
  if (k < 1.8) return new Set();
  const cx = stage.w / 2;
  const cy = stage.h / 2;
  const picked: { i: number; d: number; sx: number; sy: number; w: number }[] =
    [];
  for (const d of dots) {
    if (!d.on) continue;
    if (k < 3 && d.links < 3) continue;
    const [sx, sy] = screen(d.x, d.y);
    if (sx < 0 || sy < 0 || sx > stage.w || sy > stage.h) continue;
    picked.push({
      i: d.i,
      d: (sx - cx) ** 2 + (sy - cy) ** 2,
      sx,
      sy,
      w: d.width,
    });
  }
  picked.sort((a, b) => a.d - b.d || a.i - b.i);
  // Nearest the centre first; a label whose box (right of the dot, 20 px tall) would overlap a
  // placed one is skipped. Linear scan over at most `cap` placed boxes.
  const placed: Box[] = [];
  const out = new Set<number>();
  for (const p of picked) {
    if (out.size >= cap) break;
    const b: Box = {
      x0: p.sx + 6,
      y0: p.sy - 10,
      x1: p.sx + 6 + p.w,
      y1: p.sy + 10,
    };
    if (placed.some((q) => overlaps(q, b))) continue;
    placed.push(b);
    out.add(p.i);
  }
  return out;
}

// web/src/app/account.ts
// The signed-in member as the rail shows them: name, role and hive. One bounded read through
// ownHive (members by user_id, limit 1). Null when there is no session or no hive; the rail then
// shows no account button.
import type { SupabaseClient } from "@supabase/supabase-js";
import { ownHive, sessionUserId } from "./own-hive";

export type Account = {
  name: string;
  role: string;
  hive: string;
  initials: string;
};

export function initials(name: string): string {
  const words = name.trim().split(/\s+/).filter(Boolean);
  if (words.length === 0) return "?";
  if (words.length === 1) return words[0]!.slice(0, 2).toUpperCase();
  return (words[0]![0]! + words[1]![0]!).toUpperCase();
}

export async function loadAccount(
  db: Pick<SupabaseClient, "from" | "auth">,
): Promise<Account | null> {
  const uid = await sessionUserId(db);
  if (!uid) return null;
  const own = await ownHive(db, uid);
  if (own === null || own === "error") return null;
  return {
    name: own.displayName,
    role: own.role,
    hive: own.name,
    initials: initials(own.displayName),
  };
}

// web/src/shell/agentShort.ts
export function agentShort(lastUsedAt: string | null, now: number): string {
  if (lastUsedAt === null) return "never";
  const t = Date.parse(lastUsedAt);
  if (!Number.isFinite(t)) return "never";
  const ms = Math.max(0, now - t);
  if (ms <= 3_600_000) return "now";
  if (ms < 86_400_000) return `${Math.round(ms / 3_600_000)}h`;
  return `${Math.round(ms / 86_400_000)}d`;
}

```ts
"use client";
// Wheel/pinch zoom toward the pointer, drag to pan (4 px threshold, a drag never clicks a dot),
// double-click zooms x2, keys + = - 0 and arrows when focus is not in a text field. The world is
// sized to fit the stage at k = 1. Labels are recomputed through onSettle, 60 ms after motion.
import {
  useCallback,
  useEffect,
  useLayoutEffect,
  useRef,
  useState,
  type RefObject,
} from "react";
import { fit, frame, wheelFactor, zoomAt, type Cam } from "./camera";

export function useCamera(
  stage: RefObject<HTMLDivElement | null>,
  { enabled, onSettle }: { enabled: boolean; onSettle: () => void },
) {
  const [world, setWorld] = useState({ w: 1000, h: 760 });
  const [cam, setCam] = useState<Cam>({ k: 1, tx: 0, ty: 0 });
  const [glide, setGlide] = useState(false);
  const drag = useRef<{
    x: number;
    y: number;
    tx: number;
    ty: number;
    moved: boolean;
  } | null>(null);
  const moved = useRef(false);
  const settle = useRef<ReturnType<typeof setTimeout>>(undefined);

  const size = useCallback(() => {
    const el = stage.current;
    if (!el) return null;
    const r = el.getBoundingClientRect();
    const s = Math.min(r.width / 1000, r.height / 760);
    return { w: 1000 * s, h: 760 * s, sw: r.width, sh: r.height };
  }, [stage]);

  const set = useCallback(
    (next: Cam, smooth: boolean) => {
      setGlide(smooth);
      setCam(next);
      clearTimeout(settle.current);
      settle.current = setTimeout(onSettle, smooth ? 650 : 60);
    },
    [onSettle],
  );

  const fitView = useCallback(() => {
    const z = size();
    if (z) set(fit(z.w, z.h, z.sw, z.sh), true);
  }, [size, set]);

  useLayoutEffect(() => {
    const el = stage.current;
    if (!el) return;
    const ro = new ResizeObserver(() => {
      const z = size();
      if (!z) return;
      setWorld({ w: z.w, h: z.h });
      setGlide(false);
      setCam(fit(z.w, z.h, z.sw, z.sh));
    });
    ro.observe(el);
    return () => ro.disconnect();
  }, [stage, size]);

  useEffect(() => {
    const el = stage.current;
    if (!el || !enabled) return;
    const local = (e: { clientX: number; clientY: number }) => {
      const r = el.getBoundingClientRect();
      return [e.clientX - r.left, e.clientY - r.top] as const;
    };
    const onWheel = (e: WheelEvent) => {
      e.preventDefault();
      const [x, y] = local(e);
      setCam((c) => zoomAt(c, wheelFactor(e.deltaY, e.ctrlKey), x, y));
      setGlide(false);
      clearTimeout(settle.current);
      settle.current = setTimeout(onSettle, 60);
    };
    const onDown = (e: PointerEvent) => {
      if ((e.target as Element).closest("[data-map-overlay]")) return;
      setCam((c) => {
        drag.current = {
          x: e.clientX,
          y: e.clientY,
          tx: c.tx,
          ty: c.ty,
          moved: false,
        };
        return c;
      });
      moved.current = false;
    };
    const onMove = (e: PointerEvent) => {
      const d = drag.current;
      if (!d) return;
      const dx = e.clientX - d.x;
      const dy = e.clientY - d.y;
      if (!d.moved && Math.hypot(dx, dy) < 4) return;
      d.moved = true;
      moved.current = true;
      el.dataset.dragging = "";
      setGlide(false);
      setCam((c) => ({ ...c, tx: d.tx + dx, ty: d.ty + dy }));
    };
    const onUp = () => {
      drag.current = null;
      delete el.dataset.dragging;
      clearTimeout(settle.current);
      settle.current = setTimeout(onSettle, 60);
    };
    const onDbl = (e: MouseEvent) => {
      if ((e.target as Element).closest("[data-map-overlay]")) return;
      const [x, y] = local(e);
      setGlide(true);
      setCam((c) => zoomAt(c, 2, x, y));
    };
    el.addEventListener("wheel", onWheel, { passive: false });
    el.addEventListener("pointerdown", onDown);
    window.addEventListener("pointermove", onMove);
    window.addEventListener("pointerup", onUp);
    el.addEventListener("dblclick", onDbl);
    return () => {
      el.removeEventListener("wheel", onWheel);
      el.removeEventListener("pointerdown", onDown);
      window.removeEventListener("pointermove", onMove);
      window.removeEventListener("pointerup", onUp);
      el.removeEventListener("dblclick", onDbl);
    };
  }, [stage, enabled, onSettle]);

  const zoomBy = useCallback(
    (f: number) => {
      const z = size();
      if (z) set(zoomAt(cam, f, z.sw / 2, z.sh / 2), true);
    },
    [cam, size, set],
  );

  const frameOn = useCallback(
    (points: [number, number][]) => {
      const z = size();
      if (z && points.length) set(frame(points, z.w, z.h, z.sw, z.sh), true);
    },
    [size, set],
  );

  // A drag ends with a click event on whatever is under the pointer; dots ask this first.
  const dragged = useCallback(() => moved.current, []);

  return { cam, world, glide, zoomBy, fitView, frameOn, dragged };
}

/** How long a dot waits before gliding to a new view: outward from the centre in a wave. */
export function staggerDelay(x: number, y: number, i: number): number {
  return Math.min(450, Math.hypot(x - 500, y - 380) * 1.1) + (i % 7) * 12;

// web/src/map/sparkline.ts
import type { Activity } from "./types";

export function daySeries(act: Activity, days = 14): number[] {
  const out = new Array<number>(days).fill(0);
  for (const d of act.days)
    if (d.day >= 0 && d.day < days) out[d.day]! += d.count;
  return out;
}

/** 4 px of headroom at the top; the baseline is the bottom edge. */
export function sparkPath(series: number[], w: number, h: number) {
  const max = Math.max(1, ...series);
  const n = Math.max(1, series.length - 1);
  const pts = series.map(
    (v, i) => [(i / n) * w, h - (v / max) * (h - 10)] as const,
  );
  const line = pts
    .map(([x, y], i) => `${i ? "L" : "M"}${x.toFixed(1)} ${y.toFixed(1)}`)
    .join("");
  return { line, area: `${line}L${w} ${h}L0 ${h}Z` };
}
```
