# Eight Hand-Tuned Themes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the 18 generated themes with 8 hand-tuned ones, on flat surfaces, with a freshness colour ramp and a halo glow in Rings, without touching any layout.

**Architecture:** `themes.json` holds one flat record per theme. `tokens.ts` maps a record straight to CSS custom properties (no adjustment engine), `css.ts` writes one block per theme. Token names that components already read keep their names, so component CSS changes only where glass, scopes or the active state are involved. The canvas renderer reads the ramp, glow and hub colours from the tokens and stamps a pre-drawn halo sprite under fresh dots.

**Tech Stack:** Next.js (App Router), TypeScript, CSS Modules, Vitest, Playwright, Canvas 2D. Package `@hive-mind/web` in `web/`.

**Spec:** `docs/superpowers/specs/2026-11-24-theme-set-design.md` (section 11 lists refinements made while planning; where it differs from an earlier section, section 11 wins).

```ts
/** A straight line between two colours in OKLab: even steps in lightness, no hue detour. */
export function mixOklab(a: RGB, b: RGB, t: number): RGB {
  if (t <= 0) return a;
  if (t >= 1) return b;
  const lab = (c: RGB): [number, number, number] => {
    const o = toOklch(c);
    const rad = (o.h * Math.PI) / 180;
    return [o.l, o.c * Math.cos(rad), o.c * Math.sin(rad)];
  };
  const x = lab(a);
  const y = lab(b);
  const l = x[0] + (y[0] - x[0]) * t;
  const A = x[1] + (y[1] - x[1]) * t;
  const B = x[2] + (y[2] - x[2]) * t;
  return fromOklch(
    l,
    Math.hypot(A, B),
    ((Math.atan2(B, A) * 180) / Math.PI + 360) % 360,
  );
}
```

```ts
import data from "./themes.json";

export type Scheme = "dark" | "light";
/** The look is the theme alone; each theme is dark or light by itself. */
export type Look = { theme: string };

export const THEMES: readonly { id: string; label: string; scheme: Scheme }[] =
  data.themes.map((t) => ({
    id: t.id,
    label: t.label,
    scheme: t.scheme as Scheme,
  }));
export const DEFAULT_LOOK: Look = { theme: data.default as string };
export const LOOK_COOKIE = "hive-look";

const THEME_IDS: readonly string[] = THEMES.map((t) => t.id);

/**
 * Reads the theme id. A value from before the eight themes (`<theme>.<mode>.<reach>.<dots>`)
 * is read by its first part; anything that is not a theme falls back to the default.
 */
export function parseLook(value: string | undefined): Look {
  // A real value is a few characters; anything long is junk.
  const first =
    typeof value === "string" && value.length <= 100
      ? value.split(".")[0]
      : undefined;
  return {
    theme:
      first !== undefined && THEME_IDS.includes(first)
        ? first
        : DEFAULT_LOOK.theme,
  };
}

export function formatLook(look: Look): string {
  return look.theme;
}

export function schemeOf(theme: string): Scheme {
  return THEMES.find((t) => t.id === theme)?.scheme ?? "dark";
}
```

```ts
  fromOklch,
  mix,
  mixOklab,
  quantize,
  toOklch,
  type RGB,
} from "./colour";
import type { Scheme } from "./look";
import data from "./themes.json";

export type Tokens = Record<string, string>;
export type Theme = {
  id: string;
  label: string;
  scheme: Scheme;
  bg: string;
  card: string;
  line: string;
  text: string;
  muted: string;
  fg2: string;
  quiet: string;
  accent: string;
  onAccent: string;
  glow: string;
  hub: string;
  /** Old, middle, fresh. */
  ramp: string[];
  /** Opacity of the glow colour behind the centre of the map card. */
  tint: number;
  /** Opacity of the accent in the card's grid lines; absent means no grid. */
  grid?: number;
  activeBg: string;
  activeFg: string;
};

export const FRESH_STEPS = 8;

const LIST = data.themes as Theme[];

export function themeOf(id: string): Theme {
  const t = LIST.find((x) => x.id === id);
  if (!t) throw new Error(`unknown theme: ${id}`);
  return t;
}

/** The ramp at eight even steps, old to fresh, through the middle colour. */
export function ramp(t: Theme): RGB[] {
  const [a, b, c] = t.ramp.map(fromHex) as [RGB, RGB, RGB];
  return Array.from({ length: FRESH_STEPS }, (_, i) => {
    const u = i / (FRESH_STEPS - 1);
    return quantize(
      u < 0.5 ? mixOklab(a, b, u * 2) : mixOklab(b, c, (u - 0.5) * 2),
    );
  });
}

/**
 * The eight topic colours of Timeline and Orbit: eight hues fanned across 160 degrees around
 * the hue midway between the accent and the glow colour, alternating two lightness levels, with
 * neighbouring topics far apart in the fan. An accent with almost no chroma gets eight
 * near-greys instead. The rule is triggered by the chroma, never by the theme's name.
 */
export function topicColours(t: Theme): RGB[] {
  const light = t.scheme === "light";
  const h1 = toOklch(fromHex(t.accent));
  const h2 = toOklch(fromHex(t.glow));
  const midH = h1.h + (((h2.h - h1.h + 540) % 360) - 180) / 2;
  return [0, 4, 1, 5, 2, 6, 3, 7].map((k, i) => {
    if (h1.c < 0.03) {
      const l = Number((light ? 0.3 + k * 0.06 : 0.55 + k * 0.055).toFixed(3));
      return quantize(fromOklch(l, 0.012, Number(midH.toFixed(0))));
    }
    const l = light ? (i % 2 ? 0.5 : 0.6) : i % 2 ? 0.72 : 0.82;
    const h = Number(((midH - 80 + (k * 160) / 7 + 720) % 360).toFixed(0));
    return quantize(fromOklch(l, light ? 0.15 : 0.14, h));
  });
}

const alpha = (hex: string, a: number) => css({ ...fromHex(hex), a });

export function tokens(id: string): Tokens {
  const t = themeOf(id);
  const light = t.scheme === "light";
  const tint = `radial-gradient(circle at 50% 52%, ${alpha(t.glow, t.tint)} 0%, transparent 58%)`;
  const g = t.grid === undefined ? null : alpha(t.accent, t.grid);
  const out: Tokens = {
    "--bg": t.bg,
    "--surface": t.card,
    "--glass": t.card,
    "--glass2": t.card,
    "--panel-solid": t.card,
    // Kept so the cards' `background: var(--glow), var(--glass)` stays valid; nothing glows there.
    "--glow": "none",
    "--card-tint": g
      ? `linear-gradient(${g} 1px, transparent 1px) 0 0 / 28px 28px, linear-gradient(90deg, ${g} 1px, transparent 1px) 0 0 / 28px 28px, ${tint}`
      : tint,
    "--fg": t.text,
    "--fg2": t.fg2,
    "--muted": t.muted,
    "--quiet": t.quiet,
    "--line": t.line,
    "--line-hi": css(quantize(mix(fromHex(t.line), fromHex(t.text), 0.25))),
    "--fill": alpha(t.text, 0.06),
    "--fill-on": t.activeBg,
    "--sel": alpha(t.accent, 0.14),
    "--brand": t.accent,
    "--brand-2": t.accent,
    "--acc": t.accent,
    // One colour now; the name stays so the rules that fill with it are not edited.
    "--grad": t.accent,
    "--on-brand": t.onAccent,
    "--edge": t.line,
    "--edge-card": t.line,
    "--lift": light ? `0 1px 3px 0 ${alpha(t.text, 0.08)}` : "none",
    "--mark": t.accent,
    "--hub": t.hub,
    "--glow-colour": t.glow,
    "--halo-blend": light ? "source-over" : "lighter",
    "--active-bg": t.activeBg,
    "--active-fg": t.activeFg,
  };
  ramp(t).forEach((c, i) => {
    out[`--fresh-${i}`] = css(c);
  });
  topicColours(t).forEach((c, i) => {
    out[`--hue-${i + 1}`] = css(c);
  });
  return out;
}
```

```ts
  return 0.5 + 0.08 * Math.sin((TAU * time) / 4000);
}

const sprites = new Map<string, HTMLCanvasElement>();

/** The halo of one colour, drawn once: a radial fade on a small canvas, stamped with drawImage. */
function haloSprite(colour: string): HTMLCanvasElement {
  const hit = sprites.get(colour);
  if (hit) return hit;
  const c = document.createElement("canvas");
  c.width = c.height = HALO_PX;
  const g = c.getContext("2d")!;
  const base = parseColour(colour);
  const r = HALO_PX / 2;
  const fade = g.createRadialGradient(r, r, 0, r, r, r);
  for (const [at, share] of HALO_STOPS)
    fade.addColorStop(at, css({ ...base, a: share }));
  g.fillStyle = fade;
  g.fillRect(0, 0, HALO_PX, HALO_PX);
  // A handful of colours per theme; a theme change leaves the old ones to be collected.
  if (sprites.size > 16) sprites.clear();
  sprites.set(colour, c);
  return c;
}

/** Stamps a halo of radius `r` (in drawing units) at (x, y), at opacity `a`. */
function stampHalo(
  F: Frame,
  colour: string,
  x: number,
  y: number,
  r: number,
  a: number,
) {
  if (a <= 0 || r <= 0) return;
  const { ctx, pal } = F;
  ctx.globalAlpha = a;
  ctx.globalCompositeOperation = pal.haloBlend;
  ctx.drawImage(haloSprite(colour), x - r, y - r, 2 * r, 2 * r);
  ctx.globalCompositeOperation = "source-over";
  ctx.globalAlpha = 1;
}
```
